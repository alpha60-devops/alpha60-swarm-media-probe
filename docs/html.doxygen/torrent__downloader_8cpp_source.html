77// Returns downloaded, waits for cache flushes
78bool
79media_downloader::drain_alerts(lt::session& sesh,
80 lt::torrent_handle& /*handle*/)
81{
82 // Wait for up to 1 second for a libtorrent alert
83 bool ret(false);
84 uint flushed(0);
85 lt::alert const* a = sesh.wait_for_alert(lt::seconds(1));
86 if (a != nullptr)
87 {
88 std::vector<lt::alert*> alerts;
89 sesh.pop_alerts(&alerts);
90 for (lt::alert* alert : alerts)
91 {
92 if (lt::alert_cast<lt::torrent_removed_alert>(alert))
93 {
94 std::cout << "torrent removed " << std::endl;
95 break;
96 }
97
98 if (lt::alert_cast<lt::torrent_finished_alert>(alert))
99 {
100 std::cout << "torrent finished " << alert->message() << std::endl;
101 ret = true;
102 break;
103 }
104
105 if (lt::alert_cast<lt::torrent_error_alert>(alert))
106 {
107 std::cout << "torrent error: " << alert->message() << std::endl;
108 ret = true;
109 break;
110 }
111
112 if (lt::alert_cast<lt::cache_flushed_alert>(alert))
113 {
114 std::cout << "cache flushed ";
115 flushed++;
116 }
117
118 if (lt::alert_cast<lt::save_resume_data_alert>(alert))
119 {
120 std::cout << "save resume ";
121 flushed++;
122 }
123
124 if (lt::alert_cast<lt::save_resume_data_failed_alert>(alert))
125 {
126 std::cout << "save resume failed: " << alert->message();
127 }
128 }
129 }
130 return ret ? true : flushed == 2;

133/// Copy the first N bytes from source file to destination file
134bool
135copy_first_n_bytes(const fs::path& source, const fs::path& destination,
136 const std::int64_t num_bytes)
137{
138 std::ifstream src(source, std::ios::binary);
139 if (!src.is_open())
140 {
141 std::cerr << "[ERROR] Cannot open source file: " << source << std::endl;
142 return false;
143 }
144
145 // Read.
146 // std::int64_t buffer_size = 1024 * 1024; // 1MB buffers
147 std::int64_t buffer_size = num_bytes;
148 std::vector<char> buffer(buffer_size);
149 std::int64_t remaining = num_bytes;
150 std::int64_t total_copied = 0;
151 while (remaining > 0)
152 {
153 size_t to_read = static_cast<size_t>(std::min(remaining, buffer_size));
154 src.read(buffer.data(), to_read);
155 std::streamsize bytes_read = src.gcount();
156 if (bytes_read != 0)
157 {
158 remaining -= bytes_read;
159 total_copied += bytes_read;
160 }
161 else
162 break;
163 }
164 src.close();
165
166 // Ensure data is written to disk before returning
167 bool verifiedp = total_copied >= num_bytes;
168 if (verifiedp)
169 {
170 // Write.
171 std::ofstream dst(destination, std::ios::binary);
172 if (!dst.is_open())
173 {
174 std::cerr << "cannot create destination file: " << destination << std::endl;
175 return false;
176 }
177 dst.write(buffer.data(), total_copied);
178 dst.flush();
179 dst.close();
180
181 // Verify.
182 int fd = open(destination.string().c_str(), O_WRONLY);
183 if (fd != -1)
184 {
185 if (fsync(fd) == 0)
186 std::cout << "fsync() confirmed for .sized file";
187 else
188 std::cout << "fsync() failed for .sized file: " << strerror(errno);
189 std::cout << std::endl;
190 close(fd);
191 }
192 }
193 else
194 {
195 std::cout << "incomplete (" << total_copied <<") copied of "
196 << num_bytes << std::endl;
197 }
198
199 return verifiedp;
200}

203/// @param file_path path to the file to verify
204/// @param min_size minimum required file size (bytes)
205/// @param bytes_to_check number of bytes to read from start (default 1024)
206/// @return true if file exists, size >= min_size, and first bytes_to_check are non-zero
207bool
208verify_data_on_disk(const fs::path& file_path,
209 const std::uint64_t min_size,
210 const std::size_t bytes_to_check = 1024)
211{
212 bool result = false;
213 if (fs::exists(file_path))
214 {
215 const auto file_size = fs::file_size(file_path);
216 if (file_size >= min_size)
217 {
218 std::ifstream file(file_path, std::ios::binary);
219 if (file.is_open())
220 {
221 std::vector<char> buffer(bytes_to_check);
222 file.read(buffer.data(), bytes_to_check);
223 std::streamsize bytes_read = file.gcount();
224 file.close();
225
226 if (bytes_read > 0)
227 {
228 // Check if at least one byte is non‑zero
229 bool found_nonzero = false;
230 for (std::streamsize i = 0; i < bytes_read; ++i)
231 {
232 if (buffer[i] != 0)
233 {
234 found_nonzero = true;
235 break;
236 }
237 }
238 if (found_nonzero)
239 result = true;
240 }
241 }
242 }
243 }
244 return result;
245}

262/// Timeout loop for the downloader, active downloading until exit.
263/// @param psize is in mb
264void
265media_downloader::just_a_bit(lt::session& sesh, lt::torrent_handle& handle,
266 const time_limits& tlimits,
267 const lt::file_index_t p_index,
268 const probe_size psize)
269{
270 using namespace std;
271
272 auto [ p_mb, target_mb] = psize;
273
274 auto to_seconds = [](auto duration)
275 { return std::chrono::duration_cast<std::chrono::seconds>(duration); };
276
277 // Start timeout loop...
278 // Ends if:
279 /// 1: enough downloaded to make sized_file
280 /// 2: timeout
281 auto start_time = chrono::steady_clock::now();
282 auto last_status_time = start_time;
283
284 int64_t last_downloaded = 0;
285 auto last_rate_time = start_time;
286 double current_rate_bps = 0.0;
287 while (true)
288 {
289 // Check clock.
290 auto now = chrono::steady_clock::now();
291 auto elapsed = to_seconds(now - start_time).count();
292
293 // Get status.
294 auto status = handle.status();
295
296 // Check if the download has reached the target size and time.
297 // status.total_payload_download
298 // status.all_time_download
299 const double tdownloaded_mb = to_mb(status.total_done);
300
301 // Ask the torrent handle for the download progress of ALL files
302 std::vector<std::int64_t> file_progress;
303 handle.file_progress(file_progress);
304 std::int64_t fdownloaded = file_progress[p_index];
305 auto fdownloaded_mb = fdownloaded ? to_mb(fdownloaded) : 0;
306 if (tdownloaded_mb >= p_mb || fdownloaded_mb >= target_mb)
307 {
308 cerr << "progress (" << fdownloaded_mb << ") of " << target_mb
309 << endl;
310 break;
311 }
312
313 // Check for timeout.
314 if (elapsed > tlimits.maximum)
315 {
316 cerr << "timeout after " << tlimits.maximum << " seconds" << endl;
317 break;
318 }
319
320 // Calculate current download rate.
321 double rate_elapsed = to_seconds(now - last_rate_time).count();
322 if (rate_elapsed >= 5 && status.total_done > 0)
323 {
324 double delta_bytes = status.total_done - last_downloaded;
325 current_rate_bps = delta_bytes / rate_elapsed;
326 last_downloaded = status.total_done;
327 last_rate_time = now;
328 }
329
330 // Check if stalled.
331 if (current_rate_bps == 0 && elapsed >= tlimits.unresponsive)
332 {
333 cerr << "stalled in (" << elapsed << ") seconds" << endl;
334 break;
335 }
336
337 // Show progress every n second interval.
338 const auto status_interval = 5;
339 auto status_elapsed = to_seconds(now - last_status_time).count();
340 if (status_elapsed >= status_interval)
341 {
342 double rate_kbps = current_rate_bps / 1024.0;
343 cout << fixed << setprecision(2);
344 cout << " Peers: " << status.num_peers
345 << " | Downloaded: " << fdownloaded_mb << ", "
346 << tdownloaded_mb << " MB / " << p_mb << " MB"
347 << " | Speed: " << rate_kbps << " KB/s";
348 cout << endl;
349 last_status_time = now;
350 }
351
352 // Force a flush periodically
353 if (elapsed % 5 == 0 && elapsed > 0)
354 handle.force_reannounce();
355
356 drain_alerts(sesh);
357 this_thread::sleep_for(chrono::seconds(1));
358 }
359}

363void
364media_downloader::is_enough(lt::session& sesh, lt::torrent_handle& handle,
365 const uint max_wait)
366{
367 // Pause torrent.
368 // Disable auto_managed so the session doesn't restart it
369 using namespace std;
370 // Clear auto_managed flag (disable automatic management)
371 handle.set_flags({}, lt::torrent_flags::auto_managed);
372 handle.pause();
373
374 // Wait for it to actually stop
375 for (uint attempt = 0; attempt < max_wait; ++attempt)
376 {
377 auto status = handle.status();

379 if (status.flags & lt::torrent_flags::paused)

382 break;
383 }
384 this_thread::sleep_for(chrono::seconds(1));
385 drain_alerts(sesh);

387

389 // Request resume data (this also forces dirty blocks to disk)
390 handle.flush_cache();
391 handle.save_resume_data(lt::torrent_handle::save_info_dict);
392
393 // Session shutdown handled later.
394 double downloaded = handle.status().total_done;
395 if (downloaded != 0)
396 {
397 cout << "handle tear down: ";
398 bool donep(false);
399 for (uint i = 0; i < max_wait && !donep; ++i)
400 {
401 cout << i << ", ";
402 donep = drain_alerts(sesh, handle);
403 }

409/// Download a minimum-sized chunk of the largest media file.
410/// So that ffmpeg, mediainfo, and others can be used to determine the
411/// frame rate, frame size, audio and subtitles.
412///
413/// Download the largest file in the @parm torrent_path given as an
414/// argument, but stop at 10MB (or @param bytes_to_download) and
415/// archive the smaller sized file.
416///
417/// @param timeout_seconds the number of seconds to loop while wating for data.
418/// @param output_dir result files
419/// @param fsuffix the suffix used on the minimal media file, default ".sized"
420std::optional<fs::path>
421media_downloader::almost_nothing(const std::string& ifile,
422 const std::string& output_dir,
423 const probe_size psize,
424 const std::string fsuffix)
425{
426 // Return sized_file_path file whenever possible, as that is the small one.
427 using namespace std;
428 optional<fs::path> ret(nullopt);
429 auto [ min_fsize, max_dlsize ] = psize;
430
431 fs::path prime_file_path;
432 fs::path sized_file_path;
433 try
434 {
435 fs::create_directories(output_dir);
436
437 auto ti = make_shared<lt::torrent_info>(ifile);
438 if (ti->num_files() == 0)
439 return ret;
440
441 // Find the largest file, and prepare to download that.
442 const auto& files = ti->files();
443 lt::file_index_t largest_file_index{0};
444 int64_t max_size = -1;
445 for (int i = 0; i < ti->num_files(); ++i)
446 {
447 lt::file_index_t idx{i};
448 if (files.file_size(idx) > max_size)
449 {
450 max_size = files.file_size(idx);
451 largest_file_index = idx;
452 }
453 }
454 const double max_mb = to_mb(max_size);
455 const double target_mb = to_mb(min_fsize);
456
457 auto target_file_path = files.file_path(largest_file_index);
458 prime_file_path = fs::path(output_dir) / target_file_path;
459 sized_file_path = prime_file_path.string() + fsuffix;
460 fs::create_directories(prime_file_path.parent_path());
461
462 cout << " target file: (" << target_mb << "/" << max_mb << ")"
463 << "\t" << target_file_path << endl;
464
465 // Set up parameters.
466 lt::add_torrent_params params = { };
467 params.ti = ti;
468 params.save_path = output_dir;
469 params.flags = lt::torrent_flags_t{};
470 params.flags |= lt::torrent_flags::auto_managed;
471 //params.storage_mode = lt::storage_mode_allocate;
472 params.storage_mode = lt::storage_mode_sparse;
473
474 // Set up download priorities.
475 vector<lt::download_priority_t> priorities;
476 priorities.resize(ti->num_files(), lt::download_priority_t{0});
477 priorities[static_cast<int>(largest_file_index)] = lt::download_priority_t{7};
478 params.file_priorities = priorities;
479
480 // Start BTIH in session...
481 lt::session sesh(make_settings_pack());
482 lt::torrent_handle handle = sesh.add_torrent(move(params));
483 cout << "starting, waiting for metadata...";
484 for (int attempt = 0; attempt < 60; ++attempt)
485 {
486 if (handle.status().has_metadata)
487 break;
488 drain_alerts(sesh);
489 this_thread::sleep_for(chrono::milliseconds(500));
490 }
491 if (!handle.status().has_metadata)
492 {
493 sesh.remove_torrent(handle);
494 return ret;
495 }
496 cout << " ...metadata received." << endl;
497
498 try
499 {
500 // Loop with increasing download target sizes until prime fills in.
501 // 16, 32, 64, 128, 256, 512 (aka exponential).
502 bool serializedp(false);
503 bool stalledp(false);
504 for (uint index = 0; index < 6 && !serializedp && !stalledp; index++)
505 {
506 uint power_of_two = 1u << index;
507 uint dl_target_mb = target_mb * power_of_two;
508 just_a_bit(sesh, handle, dtlimits, largest_file_index,
509 { dl_target_mb, to_mb(max_dlsize) });
510 is_enough(sesh, handle);
511
512 auto status = handle.status();
513 if (verify_data_on_disk(prime_file_path, min_fsize))
514 serializedp = true;
515 else
516 {
517 if (status.total_done == 0 && status.download_rate == 0)
518 stalledp = true;
519 else
520 {
521 uint pfile_mb = to_mb(fs::file_size(prime_file_path));
522 cout << "try (" << index << ", "
523 << pfile_mb << " of " << dl_target_mb
524 << ") complete" << endl;
525
526 // Restart torrent: set auto_managed flag and resume
527 handle.set_flags(lt::torrent_flags::auto_managed, lt::torrent_flags::auto_managed);
528 handle.resume();
529
530 // Wait for it to actually start
531 const uint max_wait = 20;
532 for (uint attempt = 0; attempt < max_wait; ++attempt)
533 {
534 this_thread::sleep_for(chrono::seconds(1));
535 status = handle.status();
536 // Check paused flag via bitmask
537 bool readyp = !(status.flags & lt::torrent_flags::paused) && status.has_metadata;
538 bool managedp = (status.flags & lt::torrent_flags::auto_managed) != 0;
539 bool peersp = status.num_peers > 0;
540 if (readyp && managedp && peersp)
541 {
542 cout << "resumed (" << attempt << " sec), "
543 << "peers (" << status.num_peers << ")" << endl;
544 break;
545 }
546 drain_alerts(sesh);
547 }
548 }
549 }
550 }
551
552 sesh.remove_torrent(handle);
553 const uint max_wait = 10;
554 bool removedp(false);
555 for (uint i = 0; i < max_wait && !removedp; ++i)
556 {
557 cout << i << ", ";
558 removedp = drain_alerts(sesh, handle);
559 }
560 cout << endl;
561 }
562 catch (std::exception& e)
563 {
564 cout << "almost_nothing:: exception thrown during tear down ";
565 cout << endl;
566 cout << e.what();
567 cout << endl;
568 }
569
570 // Initiate session shutdown.
571 lt::session_proxy proxy = sesh.abort(); // just this is scope async
572
573 // ...but force the proxy to destroy itself right here. This line
574 // blocks this background thread until shutdown is 100% finished.
575 proxy = lt::session_proxy();
576 cout << "session done" << endl;
577 }
578 catch (const exception& e)
579 {
580 cerr << "almost_nothing: error, exception thrown " << e.what() << endl;
581 return ret;
582 }
583
584 // Settle.
585 this_thread::sleep_for(chrono::seconds(5));
586
587 // Create sized archive file from prime download file.
588 // Confirm prime_file data on disk, create sized file.
589 // Check actual file on disk for non-zero data.
590 // Create a specially-sized small file for the media archive.
591 const bool ff_created = fs::exists(prime_file_path);
592 const auto ff_size = ff_created ? fs::file_size(prime_file_path) : 0;
593 const bool ff_size_targetp = ff_size >= ulong(min_fsize);
594 if (ff_created && ff_size_targetp)
595 {
596 if (verify_data_on_disk(prime_file_path, min_fsize))
597 {
598 if (!copy_first_n_bytes(prime_file_path, sized_file_path,
599 min_fsize))
600 cerr << "fail: sized file not copied from prime file " << endl
601 << prime_file_path.string() << endl;
602 }
603 else
604 cerr << "fail: verification failed ("
605 << to_mb(ff_size) << ") in " << endl
606 << prime_file_path.string() << endl;
607 }
608
609 // Clean up
610 // Remove large file and used sized file if possible.
611 const bool cleanupp(true);
612 if (cleanupp && ff_created)
613 {
614 error_code ec;
615 if (!fs::remove(prime_file_path, ec))
616 cout << "error: failed to remove file: " << ec.message() << endl;
617 }
618
619 if (verify_data_on_disk(sized_file_path, min_fsize))
620 return sized_file_path;
621 else
622 {
623 if (ff_size == 0)
624 {
625 // Log in working directory.
626 fs::path ppath = fs::path(output_dir).parent_path().parent_path();
627 ofstream& ofno = log_suspect(ppath.lexically_normal().string());
628 ofno << ifile << endl;
629 ofno.flush();
630 }
631 return ret;
632 }
