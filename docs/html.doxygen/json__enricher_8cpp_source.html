6enrichment::compute_pipeline_metrics(const process_result& presult,

8{

10 const std::vector<fs::path>& downloaded_files = presult.downloaded_files;
11
12 pipeline_metrics metrics;
13 metrics.media_cache_file_size = mini_size;
14 metrics.btiha_size = downloaded_files.size();
15 metrics.btiha_cached_size = csize;
16
17 for (size_t i = 0; i < downloaded_files.size(); ++i)
18 {
19 const auto& path = downloaded_files[i];
20 const auto& md = (i < media_data.size()) ? media_data[i] : media_info_data();
21
22 // Check if file exists and has size
23 bool file_exists = !path.empty() && fs::exists(path);
24 uintmax_t file_size = file_exists ? fs::file_size(path) : 0;
25 if (file_size == 0)
26 {
27 metrics.btiha_unreachable_size++;
28 continue;
29 }
30
31 // Check if extraction produced usable metadata.
32 // If so, call this media object "extracted"
33
34 // Video metadata
35 bool has_vmetadata = false;
36 const bool has_vsize = md.video.width > 0 && md.video.height > 0;
37 if (has_vsize || !md.video.frame_rate.empty())
38 has_vmetadata = true;
39
40 // Audio metadata
41 bool has_ametadata = false;
42 if (md.audio.bitrate > 0 || md.audio.sampling_rate > 0)
43 has_ametadata = true;
44
45 // Format metadata
46 bool has_fmetadata = false;
47 if (md.duration > 0 || md.file_size > 0)
48 has_fmetadata = true;
49
50 // Use these individual flags to assess extraction.
51 if (has_vmetadata && has_ametadata && has_fmetadata)
52 metrics.btiha_extracted_size++;
53 else
54 metrics.btiha_partial_size++;
55 }
56
57 // Calculate percentage
58 if (metrics.btiha_size > 0)
59 {
60 double br = double(metrics.btiha_extracted_size) / metrics.btiha_size;
61 metrics.btiha_extracted_percent = br * 100.0;
62 }
63
64 return metrics;
65}

206std::string
207enrichment::escape_json_string(const std::string& str)
208{
209 std::stringstream ss;
210 for (char c : str)
211 {
212 switch (c)
213 {
214 case '"': ss << "\\\""; break;
215 case '\\': ss << "\\\\"; break;
216 case '\b': ss << "\\b"; break;
217 case '\f': ss << "\\f"; break;
218 case '\n': ss << "\\n"; break;
219 case '\r': ss << "\\r"; break;
220 case '\t': ss << "\\t"; break;
221 default: ss << c; break;
222 }
223 }
224 return ss.str();
