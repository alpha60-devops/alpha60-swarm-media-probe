1// media_cache_survey.cpp
2//
3// Linux-only C++20 survey utility for alpha60 download.cache artifacts.
4//
5// The public entry point is:
6//
7// bool survey_media_cache(
8// const std::filesystem::path& torrent_directory,
9// const std::filesystem::path& download_cache_root,
10// bool verbose = false);
11//
12// It:
13// 1. inventories *.torrent files in torrent_directory;
14// 2. recursively inventories *.sized files below download_cache_root;
15// 3. classifies each sized file by lightweight container metadata markers;
16// 4. runs MediaInfo with a configurable timeout;
17// 5. for oversized files, uses ffprobe + ffmpeg stream-copy remuxing to create
18// a bounded reduced archive and validates that archive with MediaInfo;
19// 6. atomically serializes all named lists plus per-file diagnostics to JSON.
20//
21// External runtime requirements:
22// mediainfo, ffprobe, ffmpeg
23//
24// Build example:
25// g++ -std=c++20 -O2 -Wall -Wextra -pedantic media_cache_survey.cpp -o media_cache_survey
26//
27// This file uses RapidJSON (header‑only) for all JSON operations.

184std::vector<fs::path>
185find_regular_files(const fs::path& root, std::string_view suffix, bool recursive)
186{
187 std::vector<fs::path> result;
188 std::error_code error;
189
190 const auto dopt = fs::directory_options::skip_permission_denied;
191 if (recursive)
192 {
193 fs::recursive_directory_iterator iterator(root, dopt, error);
194
195 const fs::recursive_directory_iterator end;
196 while (!error && iterator != end)
197 {
198 const fs::directory_entry& entry = *iterator;
199 std::error_code type_error;
200 const bool fpassp = entry.is_regular_file(type_error) && !type_error;
201 if (fpassp && has_suffix_case_insensitive(entry.path(), suffix))
202 result.push_back(entry.path());
203 iterator.increment(error);
204 }
205 }
206 else
207 {
208 fs::directory_iterator iterator(root, dopt, error);
209
210 const fs::directory_iterator end;
211 while (!error && iterator != end)
212 {
213 const fs::directory_entry& entry = *iterator;
214 std::error_code type_error;
215 const bool fpassp = entry.is_regular_file(type_error) && !type_error;
216 if (fpassp && has_suffix_case_insensitive(entry.path(), suffix))
217 result.push_back(entry.path());
218 iterator.increment(error);
219 }
220 }
221
222 std::sort(result.begin(), result.end());
223 return result;
224}

227std::vector<std::uint8_t>
228read_sampled_bytes(const fs::path& path,
229 std::size_t head_size, std::size_t tail_size)
230{
231 std::vector<std::uint8_t> result;
232 std::ifstream stream(path, std::ios::binary);
233
234 if (stream)
235 {
236 stream.seekg(0, std::ios::end);
237 const std::streamoff length = stream.tellg();
238 stream.seekg(0, std::ios::beg);
239
240 if (length > 0)
241 {
242 const std::size_t total = static_cast<std::size_t>(length);
243 const std::size_t head = std::min(head_size, total);
244 result.resize(head);
245 stream.read(reinterpret_cast<char*>(result.data()), static_cast<std::streamsize>(head));
246 result.resize(static_cast<std::size_t>(stream.gcount()));
247
248 if (total > head && tail_size > 0)
249 {
250 const std::size_t tail = std::min(tail_size, total - head);
251 const std::size_t old_size = result.size();
252 result.resize(old_size + tail);
253 stream.clear();
254 stream.seekg(static_cast<std::streamoff>(total - tail), std::ios::beg);
255 stream.read(reinterpret_cast<char*>(result.data() + old_size),
256 static_cast<std::streamsize>(tail));
257 result.resize(old_size + static_cast<std::size_t>(stream.gcount()));
258 }
259 }
260 }
261
262 return result;
263}

305std::string
306classify_container(const fs::path& sized_path,
307 const std::vector<std::uint8_t>& data)
308{
309 fs::path media_path = sized_path;
310 media_path.replace_extension();
311 const std::string extension = lower_copy(media_path.extension().string());
312 std::string result {"unknown"};
313
314 if (has_ebml_header(data))
315 result = extension == ".webm" ? "webm" : "matroska";
316 else if (contains_bytes(data, "ftyp") ||
317 contains_bytes(data, "moov") ||
318 contains_bytes(data, "moof"))
319 {
320 result = contains_bytes(data, "moof") ? "fragmented-mp4" :
321 extension == ".mov" ? "mov" : "mp4";
322 }
323 else if (extension == ".mkv")
324 result = "matroska";
325 else if (extension == ".avi")
326 result = "avi";
327 else if (extension == ".webm")
328 result = "webm";
329 else if (extension == ".mp4" || extension == ".m4v")
330 result = "mp4";
331 else if (extension == ".mov")
332 result = "mov";
333 else if (extension == ".ts" || extension == ".m2ts")
334 result = "mpeg-ts";
335
336 return result;
337}

371void
372append_pipe_data(int descriptor, std::string& destination, bool& open)
373{
374 std::array<char, 8192> buffer{};
375 bool continue_reading{true};
376
377 while (continue_reading && open)
378 {
379 const ssize_t count = read(descriptor, buffer.data(), buffer.size());
380 if (count > 0)
381 destination.append(buffer.data(), static_cast<std::size_t>(count));
382 else if (count == 0)
383 open = false;
384 else if (errno == EAGAIN || errno == EWOULDBLOCK || errno == EINTR)
385 continue_reading = false;
386 else
387 open = false;
388 }
389}

391command_result
392run_command(const std::vector<std::string>& arguments,
393 std::chrono::seconds timeout)
394{
395 command_result result;
396 int stdout_pipe[2]{-1, -1};
397 int stderr_pipe[2]{-1, -1};
398 const auto start = std::chrono::steady_clock::now();
399
400 if (!arguments.empty() &&
401 pipe2(stdout_pipe, O_CLOEXEC | O_NONBLOCK) == 0 &&
402 pipe2(stderr_pipe, O_CLOEXEC | O_NONBLOCK) == 0)
403 {
404 const pid_t child = fork();
405 if (child == 0)
406 {
407 setpgid(0, 0);
408 dup2(stdout_pipe[1], STDOUT_FILENO);
409 dup2(stderr_pipe[1], STDERR_FILENO);
410 close(stdout_pipe[0]);
411 close(stdout_pipe[1]);
412 close(stderr_pipe[0]);
413 close(stderr_pipe[1]);
414
415 std::vector<char*> argv;
416 argv.reserve(arguments.size() + 1);
417 for (const std::string& argument : arguments)
418 {
419 argv.push_back(const_cast<char*>(argument.c_str()));
420 }
421 argv.push_back(nullptr);
422 execvp(argv.front(), argv.data());
423 _exit(127);
424 }
425 else if (child > 0)
426 {
427 result.started = true;
428 setpgid(child, child);
429 close_fd(stdout_pipe[1]);
430 close_fd(stderr_pipe[1]);
431
432 bool stdout_open{true};
433 bool stderr_open{true};
434 bool child_running{true};
435 int status{0};
436
437 while (child_running || stdout_open || stderr_open)
438 {
439 const auto now = std::chrono::steady_clock::now();
440 if (child_running && now - start >= timeout)
441 {
442 result.timed_out = true;
443 kill(-child, SIGTERM);
444 std::this_thread::sleep_for(200ms);
445 kill(-child, SIGKILL);
446 }
447
448 std::array<pollfd, 2> poll_descriptors{{
449 {stdout_pipe[0], static_cast<short>(stdout_open ? POLLIN | POLLHUP : 0), 0},
450 {stderr_pipe[0], static_cast<short>(stderr_open ? POLLIN | POLLHUP : 0), 0}}};
451 poll(poll_descriptors.data(), poll_descriptors.size(), 50);
452
453 if (stdout_open && (poll_descriptors[0].revents & (POLLIN | POLLHUP | POLLERR)))
454 {
455 append_pipe_data(stdout_pipe[0], result.standard_output, stdout_open);
456 }
457 if (stderr_open && (poll_descriptors[1].revents & (POLLIN | POLLHUP | POLLERR)))
458 {
459 append_pipe_data(stderr_pipe[0], result.standard_error, stderr_open);
460 }
461
462 if (child_running)
463 {
464 const pid_t waited = waitpid(child, &status, WNOHANG);
465 if (waited == child)
466 {
467 child_running = false;
468 }
469 }
470 }
471
472 if (child_running)
473 {
474 waitpid(child, &status, 0);
475 }
476
477 if (WIFEXITED(status))
478 {
479 result.exit_code = WEXITSTATUS(status);
480 }
481 else if (WIFSIGNALED(status))
482 {
483 result.terminated_by_signal = true;
484 result.exit_code = 128 + WTERMSIG(status);
485 }
486 }
487 }
488
489 close_fd(stdout_pipe[0]);
490 close_fd(stdout_pipe[1]);
491 close_fd(stderr_pipe[0]);
492 close_fd(stderr_pipe[1]);
493 result.elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
494 std::chrono::steady_clock::now() - start);
495 return result;
496}

500// MediaInfo JSON validation using RapidJSON
501// -----------------------------------------------------------------------------
502bool
503mediainfo_succeeded(const command_result& result)
504{
505 if (!result.started || result.timed_out || result.exit_code != 0)
506 return false;
507
508 rapidjson::Document doc;
509 doc.Parse(result.standard_output.c_str());
510 if (doc.HasParseError() || !doc.IsObject())
511 return false;
512
513 auto media_it = doc.FindMember("media");
514 if (media_it == doc.MemberEnd() || !media_it->value.IsObject())
515 return false;
516
517 auto track_it = media_it->value.FindMember("track");
518 if (track_it == media_it->value.MemberEnd() || !track_it->value.IsArray())
519 return false;
520
521 for (const auto& track : track_it->value.GetArray())
522 {
523 if (!track.IsObject())
524 continue;
525 auto type_it = track.FindMember("@type");
526 if (type_it != track.MemberEnd() && type_it->value.IsString())
527 {
528 if (std::string_view(type_it->value.GetString(), type_it->value.GetStringLength()) == "General")
529 return true;
530 }
531 }
532 return false;
533}

549media_probe
550run_ffprobe(const fs::path& path, std::uintmax_t source_size, const survey_config& config)
551{
552 media_probe result;
553 const std::vector<std::string> arguments {
554 "ffprobe",
555 "-v", "error",
556 "-show_format",
557 "-show_streams",
558 "-of", "json",
559 path.string()};
560
561 result.command = run_command(arguments, config.ffprobe_timeout);
562
563 if (!result.command.started || result.command.timed_out || result.command.exit_code != 0)
564 return result;
565
566 rapidjson::Document doc;
567 doc.Parse(result.command.standard_output.c_str());
568 if (doc.HasParseError() || !doc.IsObject())
569 return result;
570
571 auto format_it = doc.FindMember("format");
572 if (format_it == doc.MemberEnd() || !format_it->value.IsObject())
573 return result;
574
575 const auto& format = format_it->value;
576
577 auto duration_it = format.FindMember("duration");
578 if (duration_it != format.MemberEnd() && duration_it->value.IsString())
579 {
580 try
581 {
582 result.duration_seconds = std::stod(duration_it->value.GetString());
583 }
584 catch (...) { }
585 }
586
587 auto bitrate_it = format.FindMember("bit_rate");
588 if (bitrate_it != format.MemberEnd() && bitrate_it->value.IsString())
589 {
590 try
591 {
592 result.bitrate_bits_per_second = std::stod(bitrate_it->value.GetString());
593 }
594 catch (...) {}
595 }
596
597 auto format_name_it = format.FindMember("format_name");
598 if (format_name_it != format.MemberEnd() && format_name_it->value.IsString())
599 {
600 result.format_name = format_name_it->value.GetString();
601 }
602
603 // If bitrate is missing, estimate from duration and file size.
604 if (result.duration_seconds > 0.0 && result.bitrate_bits_per_second <= 0.0)
605 {
606 result.bitrate_bits_per_second = static_cast<double>(source_size) * 8.0 / result.duration_seconds;
607 }
608
609 result.valid = (result.duration_seconds > 0.0 && result.bitrate_bits_per_second > 0.0);
610 return result;
611}

674bool
675create_redux(file_observation& observation, const fs::path& download_cache_root,
676 const survey_config& config, bool verbose)
677{
678 bool result{false};
679 auto log = [&](const std::string& msg) { if (verbose) std::cout << msg << std::endl; };
680
681 if (container_supported_for_redux(observation.detected_container))
682 {
683 log(" - Running ffprobe on " + observation.relative_path.generic_string());
684 const media_probe probe = run_ffprobe(
685 observation.source_path,
686 observation.source_size,
687 config);
688
689 if (probe.valid)
690 {
691 const fs::path redux_directory =
692 download_cache_root /
693 ".survey" /
694 "redux" /
695 stable_path_id(observation.relative_path);
696 std::error_code error;
697 fs::create_directories(redux_directory, error);
698
699 if (!error)
700 {
701 observation.redux_path = redux_directory / original_media_filename(observation.source_path);
702 fs::path temporary_path = observation.redux_path;
703 temporary_path += ".partial";
704
705 const double target_bytes =
706 static_cast<double>(config.max_archive_size) * config.archive_size_margin;
707 double target_seconds = std::min(
708 probe.duration_seconds,
709 target_bytes * 8.0 / probe.bitrate_bits_per_second);
710
711 for (std::size_t attempt = 0;
712 attempt < config.max_redux_attempts && !result;
713 ++attempt)
714 {
715 log(" - Redux attempt " + std::to_string(attempt+1) + " for " + observation.relative_path.generic_string());
716 fs::remove(temporary_path, error);
717 error.clear();
718 const command_result ffmpeg = run_command(
719 make_ffmpeg_arguments(
720 observation.source_path,
721 temporary_path,
722 observation.detected_container,
723 target_seconds),
724 config.ffmpeg_timeout);
725
726 if (ffmpeg.started &&
727 !ffmpeg.timed_out &&
728 ffmpeg.exit_code == 0 &&
729 fs::is_regular_file(temporary_path, error))
730 {
731 const std::uintmax_t output_size = fs::file_size(temporary_path, error);
732 if (!error && output_size > 0 && output_size <= config.max_archive_size)
733 {
734 fs::rename(temporary_path, observation.redux_path, error);
735 if (!error)
736 {
737 observation.redux_created = true;
738 observation.redux_size = output_size;
739 const std::vector<std::uint8_t> bytes = read_sampled_bytes(
740 observation.redux_path,
741 8ULL * 1024ULL * 1024ULL,
