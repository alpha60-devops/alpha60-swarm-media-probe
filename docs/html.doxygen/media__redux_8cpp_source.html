57void read_available(int descriptor, std::string& destination, bool& open)
58{
59 std::array<char, 8192> buffer{};
60 bool reading{true};
61
62 while (reading && open)
63 {
64 const ssize_t count = read(descriptor, buffer.data(), buffer.size());
65 if (count > 0)
66 destination.append(buffer.data(), static_cast<std::size_t>(count));
67 else if (count == 0)
68 open = false;
69 else if (errno == EAGAIN || errno == EWOULDBLOCK || errno == EINTR)
70 reading = false;
71 else
72 open = false;
73 }
74}

76process_result run_process(
77 const std::vector<std::string>& arguments,
78 std::chrono::seconds timeout)
79{
80 process_result result;
81 int stdout_pipe[2]{-1, -1};
82 int stderr_pipe[2]{-1, -1};
83
84 const bool pipes_ready =
85 !arguments.empty() &&
86 pipe2(stdout_pipe, O_CLOEXEC | O_NONBLOCK) == 0 &&
87 pipe2(stderr_pipe, O_CLOEXEC | O_NONBLOCK) == 0;
88
89 if (pipes_ready)
90 {
91 const auto started_at = std::chrono::steady_clock::now();
92 const pid_t child = fork();
93
94 if (child == 0)
95 {
96 setpgid(0, 0);
97 dup2(stdout_pipe[1], STDOUT_FILENO);
98 dup2(stderr_pipe[1], STDERR_FILENO);
99 close(stdout_pipe[0]);
100 close(stdout_pipe[1]);
101 close(stderr_pipe[0]);
102 close(stderr_pipe[1]);
103
104 std::vector<char*> argv;
105 argv.reserve(arguments.size() + 1);
106 for (const std::string& argument : arguments)
107 argv.push_back(const_cast<char*>(argument.c_str()));
108 argv.push_back(nullptr);
109
110 execvp(argv.front(), argv.data());
111 _exit(127);
112 }
113 else if (child > 0)
114 {
115 result.started = true;
116 setpgid(child, child);
117 close_descriptor(stdout_pipe[1]);
118 close_descriptor(stderr_pipe[1]);
119
120 bool stdout_open{true};
121 bool stderr_open{true};
122 bool child_running{true};
123 bool termination_sent{false};
124 int status{0};
125
126 while (child_running || stdout_open || stderr_open)
127 {
128 if (child_running && !termination_sent &&
129 std::chrono::steady_clock::now() - started_at >= timeout)
130 {
131 result.timed_out = true;
132 termination_sent = true;
133 kill(-child, SIGTERM);
134 std::this_thread::sleep_for(200ms);
135 kill(-child, SIGKILL);
136 }
137
138 std::array<pollfd, 2> descriptors{{
139 {stdout_pipe[0], static_cast<short>(stdout_open ? POLLIN | POLLHUP : 0), 0},
140 {stderr_pipe[0], static_cast<short>(stderr_open ? POLLIN | POLLHUP : 0), 0}}};
141
142 poll(descriptors.data(), descriptors.size(), 50);
143
144 if (stdout_open &&
145 (descriptors[0].revents & (POLLIN | POLLHUP | POLLERR)))
146 read_available(stdout_pipe[0], result.standard_output, stdout_open);
147 if (stderr_open &&
148 (descriptors[1].revents & (POLLIN | POLLHUP | POLLERR)))
149 read_available(stderr_pipe[0], result.standard_error, stderr_open);
150
151 if (child_running && waitpid(child, &status, WNOHANG) == child)
152 child_running = false;
153 }
154
155 if (WIFEXITED(status))
156 result.exit_code = WEXITSTATUS(status);
157 else if (WIFSIGNALED(status))
158 result.exit_code = 128 + WTERMSIG(status);
159 }
160 }
161
162 close_descriptor(stdout_pipe[0]);
163 close_descriptor(stdout_pipe[1]);
164 close_descriptor(stderr_pipe[0]);
165 close_descriptor(stderr_pipe[1]);
166 return result;
167}

179std::optional<redux_format> format_for(const fs::path& input)
180{
181 const std::string extension = lower_copy(input.extension().string());
182 std::optional<redux_format> result;
183
184 if (extension == ".mkv")
185 result = redux_format{"matroska", ".mkv", false};
186 else if (extension == ".webm")
187 result = redux_format{"webm", ".webm", false};
188 else if (extension == ".mp4" || extension == ".m4v")
189 result = redux_format{"mp4", ".mp4", true};
190 else if (extension == ".mov")
191 result = redux_format{"mov", ".mov", true};
192 else if (extension == ".avi")
193 result = redux_format{"avi", ".avi", false};
194 else if (extension == ".ts" || extension == ".m2ts")
195 result = redux_format{"mpegts", ".ts", false};
196
197 return result;
198}
