92bool MediaInfoExtractor::parse_ffprobe_output(const std::string& json_output, MediaInfoData& data) {
93 rapidjson::Document doc;
94 if (doc.Parse(json_output.c_str()).HasParseError()) {
95 std::cerr << " [FFprobe] JSON parse error" << std::endl;
96 return false;
97 }
98
99 if (!doc.HasMember("streams") || !doc["streams"].IsArray()) {
100 return false;
101 }
102
103 const auto& streams = doc["streams"];
104 std::vector<std::string> deduced_audio_langs;
105 std::vector<std::string> deduced_subtitle_langs;
106
107 for (rapidjson::SizeType i = 0; i < streams.Size(); ++i) {
108 const auto& stream = streams[i];
109 if (!stream.HasMember("codec_type") || !stream["codec_type"].IsString()) continue;
110
111 std::string codec_type = stream["codec_type"].GetString();
112 std::string lang = "";
113
114 if (stream.HasMember("tags") && stream["tags"].IsObject()) {
115 const auto& tags = stream["tags"];
116 if (tags.HasMember("language") && tags["language"].IsString()) {
117 lang = tags["language"].GetString();
118 }
119 }
120
121 if (!lang.empty() && lang != "und") { // Ignore 'undefined' language tags
122 if (codec_type == "audio") {
123 if (std::find(deduced_audio_langs.begin(), deduced_audio_langs.end(), lang) == deduced_audio_langs.end()) {
124 deduced_audio_langs.push_back(lang);
125 }
126 } else if (codec_type == "subtitle") {
127 if (std::find(deduced_subtitle_langs.begin(), deduced_subtitle_langs.end(), lang) == deduced_subtitle_langs.end()) {
128 deduced_subtitle_langs.push_back(lang);
129 }
130 }
131 }
132 }
133
134 // Override MediaInfo data if FFprobe successfully deduced concrete tags
135 if (!deduced_audio_langs.empty()) {
136 data.audio.languages = deduced_audio_langs;
137 }
138 if (!deduced_subtitle_langs.empty()) {
139 data.subtitle.languages = deduced_subtitle_langs;
140 }
141
142 return true;
143}

145bool MediaInfoExtractor::parse_json_output(const std::string& json_output, MediaInfoData& data) {
146 rapidjson::Document doc;
147 if (doc.Parse(json_output.c_str()).HasParseError()) {
148 std::cerr << " [MediaInfo] JSON parse error at offset " << doc.GetErrorOffset() << std::endl;
149 return false;
150 }
151
152 if (!doc.HasMember("media") || !doc["media"].IsObject()) {
153 std::cerr << " [MediaInfo] JSON missing 'media' object" << std::endl;
154 return false;
155 }
156
157 const auto& media = doc["media"];
158 if (!media.HasMember("track") || !media["track"].IsArray()) {
159 std::cerr << " [MediaInfo] JSON missing 'track' array" << std::endl;
160 return false;
161 }
162
163 const auto& tracks = media["track"];
164
165 // Initialize defaults
166 data.file_size = 0;
167 data.duration = 0.0;
168 data.overall_bit_rate = 0;
169 data.video.bitrate = 0;
170 data.video.width = 0;
171 data.video.height = 0;
172 data.audio.bitrate = 0;
173 data.audio.sampling_rate = 0;
174 data.audio.bit_depth = 0;
175 data.subtitle.format = "";
176
177 int text_track_count = 0;
178
179 // Parse each track
180 for (rapidjson::SizeType i = 0; i < tracks.Size(); ++i) {
181 const auto& track = tracks[i];
182 if (!track.HasMember("@type") || !track["@type"].IsString()) continue;
183
184 std::string track_type = track["@type"].GetString();
185
186 if (track_type == "General") {
187 data.originating_source_medium_id = extract_string_value(track, "OriginalSourceMedium_ID");
188 data.originating_source_form = extract_string_value(track, "OriginalSourceForm");
189 data.originating_network_name = extract_string_value(track, "OriginalNetworkName");
190 data.format_commercial_if_any = extract_string_value(track, "Format_Commercial_IfAny");
191 data.file_size = extract_int64_value(track, "FileSize");
192 data.duration = extract_double_value(track, "Duration");
193 data.overall_bit_rate = extract_int64_value(track, "OverallBitRate");
194 data.domain = extract_string_value(track, "Domain");
195 data.collection = extract_string_value(track, "Collection");
196 data.season = extract_string_value(track, "Season");
197 data.distributed_by = extract_string_value(track, "DistributedBy");
198 data.genre = extract_string_value(track, "Genre");
199 data.content_type = extract_string_value(track, "ContentType");
200 data.owner = extract_string_value(track, "Owner");
201 data.country = extract_string_value(track, "Country");
202 data.comment = extract_string_value(track, "Comment");
203 data.language = extract_string_value(track, "Language");
204 data.video_creation_metadata = extract_string_value(track, "Encoded_Date");
205 if (data.video_creation_metadata.empty()) {
206 data.video_creation_metadata = extract_string_value(track, "Tagged_Date");
207 }
208 }
209 else if (track_type == "Video") {
210 data.video.codec_id = extract_string_value(track, "CodecID");
211 data.video.codec_version = extract_string_value(track, "Encoded_Library");
212 if (data.video.codec_version.empty()) {

216 data.video.bitrate = extract_int64_value(track, "BitRate");
217 data.video.width = static_cast<int>(extract_int64_value(track, "Width"));
218 data.video.height = static_cast<int>(extract_int64_value(track, "Height"));
219 data.video.frame_rate = extract_string_value(track, "FrameRate");
220 // Extract color primaries (British spelling from MediaInfo)
221 data.video.color_primaries = extract_string_value(track, "colour_primaries");
222 }
223 else if (track_type == "Audio") {
224 if (data.audio.codec.empty()) {
225 data.audio.codec = extract_string_value(track, "CodecID");
226 data.audio.codec_version = extract_string_value(track, "Format_Commercial");
227 data.audio.bitrate = extract_int64_value(track, "BitRate");
228 data.audio.sampling_rate = static_cast<int>(extract_int64_value(track, "SamplingRate"));
229 data.audio.channels = extract_string_value(track, "Channels");
230 data.audio.bit_depth = static_cast<int>(extract_int64_value(track, "BitDepth"));
231 }
232 }
233 else if (track_type == "Text") {
234 text_track_count++;
235 std::string lang = extract_string_value(track, "Language");
236 if (lang.empty()) {
237 lang = "text #" + std::to_string(text_track_count);
238 }
239 if (std::find(data.subtitle.languages.begin(), data.subtitle.languages.end(), lang) == data.subtitle.languages.end()) {
240 data.subtitle.languages.push_back(lang);

251// Helper function to interpret color_primaries values for human-readable output
252std::string MediaInfoExtractor::interpret_color_primaries(const std::string& value)
253{
254 if (value.empty()) return "unknown";
255 if (value == "BT.709") return "HD (Rec.709)";
256 if (value == "BT.2020") return "UHD/HDR (Rec.2020)";
257 if (value == "BT.470 M") return "Black & White / Monochrome (NTSC)";
258 if (value == "BT.470 B,G") return "Black & White / Monochrome (PAL/SECAM)";
259 if (value == "SMPTE 170M") return "SD (NTSC)";
260 if (value == "SMPTE 240M") return "HD (SMPTE 240M)";
261 if (value == "film") return "Film / Cinematic";
262 if (value == "EBU Tech. 3213") return "SD (PAL)";
263 return value; // Return raw value if not in known list
264}
