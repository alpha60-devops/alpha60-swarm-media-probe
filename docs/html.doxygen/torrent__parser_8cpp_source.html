67std::string torrent_parser::compute_info_hash(const std::vector<char>& torrent_data) {
68 lt::error_code ec;
69 lt::bdecode_node node = lt::bdecode(
70 lt::span<const char>(torrent_data.data(), torrent_data.size()), ec);
71 if (ec) return "";

74 if (!info) return "";
